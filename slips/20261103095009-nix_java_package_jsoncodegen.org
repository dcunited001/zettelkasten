stdenv.mkDerivation (finalAttrs: {
  pname = "jsoncodegen";
  # name = "jsoncodegen";
  version = "0.12.0"; # TODO this needs to be extracted from gradle.properties or something (updater requires it)
  commit = "b5c69b0385ea73dba5a9f0bde9a69908e078c425";

  src = fetchFromGitHub {
    owner = "schlothauer-wauer";
    repo = "jsoncodegen";
    # tag = "v${finalAttrs.version}";
    rev = "b5c69b0385ea73dba5a9f0bde9a69908e078c425";
    hash = "sha256-6s0qslk/V0koDABBBKzoHFaRBpEkWRKAoFmQO3Iq0kg=";
  };

  nativeBuildInputs = [
    gradle
    jdk21
    makeWrapper
  ];

  # this is required for using mitm-cache on Darwin
  __darwinAllowLocalNetworking = true;

  # ---------------------------------------------
  # build tasks
  gradleFlags = [ "-Dfile.encoding=utf-8" ];

  # defaults to "assemble"
  gradleBuildTask = "buildRelease";
  gradleUpdateTask = "dependenciesToLibDir";
  # gradleCheckTask

  # will run the gradleCheckTask (defaults to "test")
  doCheck = true;

  installPhase = ''
    ls -al release
    ls -al release/*
    mkdir -p $out/{bin,share/jsoncodegen}
    # mkdir -p $out/share/jsoncodegen/{conf,lib}
    cp -r release $out/share/jsoncodegen

    makeWrapper ${lib.getExe jre} $out/bin/jsoncodegen \
      --add-flags "-jar $out/share/jsoncodegen/"

    # cp ${finalAttrs.src}/pdftk.1 $out/share/man/man1
  '';

  mitmCache = gradle.fetchDeps {
    # inherit (finalAttrs) pname;
    pkg = finalAttrs.finalPackage;
    data = ./deps.json;
  };

  passthru.updateScript = writeShellScript "update-jsoncodegen" ''
    ${lib.getExe nix-update} jsoncodegen

    $(nix build --no-link --print-out-paths .#jsoncodegen.mitmCache.updateScript)
  '';

  meta = {
    homepage = "https://github.com/schlothauer-wauer/jsoncodegen";
    description = "Groovy based generation tasks from JSON schema";
    license = lib.licenses.mit;
    sourceProvenance = with lib.sourceTypes; [
      fromSource
      binaryBytecode
    ];
    inherit (jre.meta) platforms;
    # maintainers = with lib.maintainers; [ msgilligan ];
  };
})
