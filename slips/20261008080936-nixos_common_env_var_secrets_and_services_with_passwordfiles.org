# rg --iglob='!*.py' '[A-Z_]{8}_?(TOKEN|KEY|PASS)_?[A-Z]+'
# rg --iglob='*.py' '[A-Z_]{8}_?(TOKEN|KEY|PASS)_?[A-Z]+'

dd=(ansible-collections ansible talos dcim hashicorp ~/forge/omarchy crypto/{smallstep,sops} x.files)
gn=(guix nixos x.files) # about half the results

rgx='[A-Z][A-Z_]{2,}_?(TOKEN|KEY|PASS|SSH|JWT|TLS)_?[A-Z_0-9]*'
rg -INo. --iglob='.env*' --iglob='*.{toml,yaml,yml,rst,md,mdx}' "$rgx" ${dd[@]} \
    | tee -ap >(rg -INo. --iglob='*.{scm,nix}' "$rgx" ${gn[@]}) \
    | grep -vE '[0-9]+' | sort | uniq
