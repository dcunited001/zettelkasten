#+begin_src sh :var q=jqCoredumps
export q;
declare -a dumps
dumps=()
while read -a recs; do
    pid=${recs[0]}; prog=${recs[1]};
    dump="$stamp-$pid-$prog.dump"
    dumps+=($dump)
done <<< $(coredumpctl list  --since "160 minutes ago" --json=short | jq -rj "$q")

for dump in ${dumps[@]}; do
    coredumpctl dump "$pid" --output="$dump"
done
#+end_src
