** Passing Current =TODO= item to OBS

it's a start...

#+begin_src emacs-lisp
(defun dc/obs-org-entry-format ()
  (let* ((todo (org-entry-get (point) "TODO"))
         (p (org-entry-get (point) "PRIORITY"))
         (priority (format "[#%s] " p))
         (item (org-entry-get (point) "ITEM")))
    (format "%s: %s %s" todo priority item)))

(defun dc/obs-org-entry-print ()
  (interactive)
  (message
   (dc/obs-org-entry-format)))
#+end_src
